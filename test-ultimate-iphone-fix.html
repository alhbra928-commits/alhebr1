🔧 التفاصيل التقنية

كيف يعمل: 

const lockHeader = () => {
  const rect = header.getBoundingClientRect();

  // إذا تحرك الهيدر حتى 1px
  if (rect.top !== 0) {
    // أعده فوراً!
    header.style.top = '0px';
    header.style.transform = 'translate3d(0,0,0)';
    fixCount++; // عد التصحيحات
  }

  frameCount++; // عد الإطارات

  // استمر في المراقبة (60fps)
  requestAnimationFrame(lockHeader);
};

// ابدأ المراقبة المستمرة
lockHeader();
