📋 استعلام تشخيصي شامل:
═══════════════════════

SELECT
    'Settings Count' as metric,
    COUNT(*)::text as value
FROM activity_ticker_settings

UNION ALL

SELECT
    'Real Activities (Visible)',
    COUNT(*)::text
FROM platform_activities
WHERE is_visible = true

UNION ALL

SELECT
    'Simulated Activities (Active)',
    COUNT(*)::text
FROM simulated_activities
WHERE is_active = true

UNION ALL

SELECT
    'Settings Mode',
    mode
FROM activity_ticker_settings
LIMIT 1

UNION ALL

SELECT
    'Simulation Enabled',
    CASE WHEN simulation_enabled THEN 'YES' ELSE 'NO' END
FROM activity_ticker_settings
LIMIT 1

UNION ALL

SELECT
    'Real Enabled',
    CASE WHEN real_enabled THEN 'YES' ELSE 'NO' END
FROM activity_ticker_settings
LIMIT 1;

│ Settings Mode               │ hybrid  │
│ Simulation Enabled          │ YES     │
