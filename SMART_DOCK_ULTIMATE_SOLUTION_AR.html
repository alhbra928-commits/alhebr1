⚙️ التفاصيل التقنية

1️⃣ إنشاء Container منفصل

// إنشاء حاوية خارج React DOM تماماً const container = document.createElement('div'); container.id = 'smart-bottom-dock-portal'; container.setAttribute('data-dock', 'true'); // إضافة مباشرة لـ body (خارج #root) document.body.appendChild(container);

2️⃣ تثبيت Position و Z-index

#smart-bottom-dock-portal { position: fixed !important; bottom: 0 !important; z-index: 2147483647 !important; /* Max 32-bit integer */ /* Force GPU acceleration */ transform: translate3d(0, 0, 0) !important; -webkit-backface-visibility: hidden !important; will-change: transform, opacity !important; }

3️⃣ منع Safari من التلاعب

// منع bounce scrolling const preventBounce = (e: TouchEvent) => { const target = e.target as HTMLElement; if (target.closest('[data-dock="true"]')) { e.preventDefault(); } }; // منع viewport resize const preventResize = () => { const viewport = document.querySelector('meta[name="viewport"]'); if (viewport) { viewport.setAttribute('content', 'width=device-width, initial-scale=1.0, ' + 'maximum-scale=1.0, user-scalable=no, viewport-fit=cover' ); } };

4️⃣ iOS Safe Area Support

.smart-dock-content { padding-bottom: env(safe-area-inset-bottom, 20px) !important; padding-bottom: max(env(safe-area-inset-bottom), 20px) !important; padding-left: env(safe-area-inset-left, 0px) !important; padding-right: env(safe-area-inset-right, 0px) !important; }

5️⃣ Overlay Interaction Layer

.smart-dock-overlay { position: fixed !important; bottom: 0 !important; height: max(90px, calc(90px + env(safe-area-inset-bottom))) !important; z-index: 2147483646 !important; /* تحت Dock مباشرة */ pointer-events: none !important; }
