🔍 التحقق من قاعدة البيانات:
════════════════════════════

-- في Supabase SQL Editor
SELECT
    activity_title_ar,
    icon,
    priority,
    is_visible,
    created_at
FROM platform_activities
WHERE is_visible = true
ORDER BY created_at DESC
LIMIT 10;
