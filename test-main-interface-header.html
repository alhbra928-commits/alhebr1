🎨 التقنيات المستخدمة

position: fixed - لتثبيت الهيدر في مكانه

env(safe-area-inset-*) - لدعم النوتش والحواف

overflow: auto - لجعل المحتوى قابل للتمرير

WebkitOverflowScrolling: touch - للتمرير السلس

backdrop-filter: blur(20px) - للتأثير الزجاجي

transform: translate3d(0,0,0) - لتسريع GPU

willChange: transform - للأداء المحسّن
