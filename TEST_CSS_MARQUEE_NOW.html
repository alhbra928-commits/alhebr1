CSS Animation:
/* الأنيميشن الأساسي */
@keyframes marquee {
0% { transform: translateX(0); }
100% { transform: translateX(-50%); }
}

/* تطبيق الأنيميشن */
.marquee-track {
animation: marquee 14s linear infinite;
will-change: transform;
}
